Q1. Write a C program to enter and print all elements of an array using a pointer.

#include <stdio.h> #include <conio.h> #include <dos.h> void main() { int a[5],*p,i; clrscr(); p=a; for(i=0;i<5;i++) { printf("Enter number at location %d: ",i+1); scanf("%d",p+i); } printf("\nArray elements:\n"); for(i=0;i<5;i++) { printf("%d ",*(p+i)); } delay(2000); }

Q2. Write a program to swap two numbers using pointers.

#include <stdio.h> #include <conio.h> #include <dos.h> void main() { int a,b,temp; int *p1,*p2; clrscr(); printf("Enter first number: "); scanf("%d",&a); printf("Enter second number: "); scanf("%d",&b); p1=&a; p2=&b; temp=*p1; *p1=*p2; *p2=temp; printf("\nAfter swapping:\n"); printf("First number = %d\n",a); printf("Second number = %d",b); delay(2000); }
