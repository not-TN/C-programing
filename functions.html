Q2.Write a program to enter 10 number from the user add the number and display the answer ?

#include <stdio.h> #include <conio.h> #include <dos.h> void add10(int *ptr1) { int i, sum = 0; printf("Enter 10 numbers:\n"); for(i = 1; i <= 10; i++) { printf("Enter number %d: ", i); scanf("%d", ptr1); sum += *ptr1; } printf("\nTotal Sum of 10 numbers = %d\n", sum); } void main() { int num1; clrscr(); add10(&num1); delay(3000); }

Q4.Making following program using local and global variable ? Factorial , Reverse , Palindrome , Prime numbers, Armstrong , Fibonacci

#include <stdio.h> #include <conio.h> #include <dos.h> int num; void factorial() { int i; long int fact=1; for(i=1;i<=num;i++) { fact=fact*i; } printf("Factorial = %ld",fact); } void reverse() { int a=num, rev=0; while(a>0) { rev=rev*10+a%10; a=a/10; } printf("Reverse = %d",rev); } void palindrome() { int a=num, rev=0; while(a>0) { rev=rev*10+a%10; a=a/10; } if(num==rev) printf("Palindrome"); else printf("Not Palindrome"); } void prime() { int i,flag=0; for(i=2;i<num;i++) { if(num%i==0) { flag=1; break; } } if(num<=1 || flag==1) printf("Not Prime"); else printf("Prime"); } void armstrong() { int a,b,c,sum; a=num%10; b=(num/10)%10; c=num/100; sum=a*a*a+b*b*b+c*c*c; if(num>=100 && num<=999 && sum==num) printf("Armstrong"); else printf("Not Armstrong"); } void fibonacci() { int i,a=0,b=1,c; for(i=1;i<=num;i++) { printf("%d ",a); c=a+b; a=b; b=c; } } void main() { int choice; clrscr(); printf("1. Factorial\n"); printf("2. Reverse\n"); printf("3. Palindrome\n"); printf("4. Prime\n"); printf("5. Armstrong\n"); printf("6. Fibonacci\n"); printf("\nEnter your choice: "); scanf("%d",&choice); if(choice>=1 && choice<=6) { printf("Enter a number: "); scanf("%d",&num); switch(choice) { case 1: if(num<0) printf("Factorial not defined"); else factorial(); break; case 2: reverse(); break; case 3: palindrome(); break; case 4: prime(); break; case 5: armstrong(); break; case 6: fibonacci(); break; } } else printf("Invalid choice"); delay(2000); }

Q5.Swap numbers using Call by address ?
