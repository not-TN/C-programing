Enter the string from user and copy reverse order into another string ?

#include<stdio.h> #include<conio.h> #include<dos.h> void main() { char a[20], b[20]; int i, j=0, count=0; clrscr(); printf("Enter a string: "); gets(a); for(i=0; a[i]!='\0'; i++) { count++; } for(i=count-1; i>=0; i--) { b[j]=a[i]; j++; } b[j]='\0'; printf("Reverse: %s", b); delay(2000); }
