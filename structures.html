Create structure of employee which there is emp.ID, name, designation of 50 employees from user and print name character with emp.ID in tabular form ?

#include <stdio.h> #include <conio.h> #include <dos.h> struct employee { int id; char name[20]; char designation[20]; }; void main() { struct employee emp[50]; int i,j; clrscr(); for(i=0;i<50;i++) { printf("\nEnter employee %d details:\n",i+1); printf("Enter ID: "); scanf("%d",&emp[i].id); printf("Enter name: "); gets(emp[i].name); printf("Enter designation: "); gets(emp[i].designation); } printf("\nID\tName Character\n"); for(i=0;i<50;i++) { for(j=0;j<strlen(emp[i].name);j++) { printf("%d\t%c\n",emp[i].id,emp[i].name[j]); } } delay(2000); }

Enter student details like roll name, fee, DOB and print all the detail by ascending order of roll number ?

#include <stdio.h> #include <conio.h> #include <dos.h> struct student { int roll; char name[20]; float fee; char dob[20]; }; void main() { struct student st[50], temp; int i,j,n; clrscr(); printf("Enter number of students (1-50): "); scanf("%d",&n); if(n<1 || n>50) { printf("Invalid number of students"); } else { for(i=0;i<n;i++) { printf("\nEnter student %d details:\n",i+1); printf("Enter roll number: "); scanf("%d",&st[i].roll); printf("Enter name: "); gets(st[i].name); printf("Enter fee: "); scanf("%f",&st[i].fee); printf("Enter DOB (DD/MM/YYYY): "); gets(st[i].dob); } for(i=0;i<n-1;i++) { for(j=i+1;j<n;j++) { if(st[i].roll>st[j].roll) { temp=st[i]; st[i]=st[j]; st[j]=temp; } } } printf("\nRoll\tName\tFee\tDOB\n"); for(i=0;i<n;i++) { printf("%d\t%s\t%f\t%s\n", st[i].roll,st[i].name,st[i].fee,st[i].dob); } } delay(2000); }
